cmake_minimum_required(VERSION 3.21)
project(AstraShell VERSION 0.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick Qml)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(AstraShell src/main.cpp src/SystemBridge.cpp src/SystemBridge.h src/ShellController.cpp src/ShellController.h)
qt_add_qml_module(AstraShell URI AstraShell VERSION 1.0 QML_FILES qml/Main.qml)
target_link_libraries(AstraShell PRIVATE Qt6::Quick Qt6::Qml)
if(WIN32)
 target_link_libraries(AstraShell PRIVATE user32)
endif()
set_target_properties(AstraShell PROPERTIES WIN32_EXECUTABLE TRUE)
install(TARGETS AstraShell RUNTIME DESTINATION bin)
